## Layout System

Layouts are specialized components that wrap page content and provide consistent structure for different content types. All layouts are located in the `layouts/` directory.

## PostLayout

The default blog post layout with full features including author info, tags, comments, and post navigation.

**Location**: `layouts/PostLayout.tsx`

### Interface

```tsx
interface LayoutProps {
  content: CoreContent<Blog>
  authorDetails: CoreContent<Authors>[]
  next?: { path: string; title: string }
  prev?: { path: string; title: string }
  children: ReactNode
}
```

## ListLayoutWithTags

Blog listing page with sidebar tag filtering.

**Location**: `layouts/ListLayoutWithTags.tsx`

### Interface

```tsx
interface ListLayoutProps {
  posts: CoreContent<Blog>[]
  title: string
  initialDisplayPosts?: CoreContent<Blog>[]
  pagination?: PaginationProps
}

interface PaginationProps {
  totalPages: number
  currentPage: number
}
```

### Features

<CardGroup cols={2}>
  <Card title="Tag Sidebar" icon="sidebar">
    Sticky sidebar with all tags and post counts
  </Card>
  <Card title="Post List" icon="list">
    Chronological list of blog posts with summaries
  </Card>
  <Card title="Pagination" icon="arrows-left-right">
    Previous/Next navigation for large post lists
  </Card>
  <Card title="Active Filter" icon="filter">
    Highlights currently selected tag
  </Card>
</CardGroup>

### Component Structure

```tsx
export default function ListLayoutWithTags({
  posts,
  title,
  initialDisplayPosts = [],
  pagination,
}: ListLayoutProps) {
  const pathname = usePathname()
  const tagCounts = tagData as Record<string, number>
  const sortedTags = Object.keys(tagCounts).sort((a, b) => 
    tagCounts[b] - tagCounts[a]
  )

  return (
    <div className="flex sm:space-x-24">
      {/* Tag Sidebar */}
      <div className="hidden sm:flex min-w-[280px] max-w-[280px]">
        <div className="px-6 py-4">
          {pathname.startsWith('/blog') ? (
            <h3 className="font-bold uppercase text-primary-500">
              All Posts
            </h3>
          ) : (
            <Link href="/blog">All Posts</Link>
          )}
          <ul>
            {sortedTags.map((t) => (
              <li key={t} className="my-3">
                <Link href={`/tags/${slug(t)}`}>
                  {`${t} (${tagCounts[t]})`}
                </Link>
              </li>
            ))}
          </ul>
        </div>
      </div>
      
      {/* Post List */}
      <div>
        <ul>
          {displayPosts.map((post) => (
            <li key={post.path} className="py-5">
              <article>
                <time dateTime={post.date}>
                  {formatDate(post.date)}
                </time>
                <h2>{post.title}</h2>
                <div>{post.tags?.map((tag) => <Tag text={tag} />)}</div>
                <p>{post.summary}</p>
              </article>
            </li>
          ))}
        </ul>
        {pagination && <Pagination {...pagination} />}
      </div>
    </div>
  )
}
```

## AuthorLayout

Layout for author profile pages.

**Location**: `layouts/AuthorLayout.tsx`

### Interface

```tsx
interface Props {
  children: ReactNode
  content: Omit<Authors, '_id' | '_raw' | 'body'>
}
```

### Structure

```tsx
export default function AuthorLayout({ children, content }: Props) {
  const { name, avatar, occupation, company, email, twitter, linkedin, github } = content

  return (
    <div className="divide-y divide-gray-200 dark:divide-gray-700">
      <div className="space-y-2 pb-8 pt-6">
        <h1 className="text-3xl font-extrabold">About</h1>
      </div>
      <div className="items-start xl:grid xl:grid-cols-3 xl:gap-x-8">
        {/* Avatar Column */}
        <div className="flex flex-col items-center pt-8">
          <Image
            src={avatar}
            alt="avatar"
            width={192}
            height={192}
            className="h-48 w-48 rounded-full"
          />
          <h3 className="pb-2 pt-4 text-2xl font-bold">{name}</h3>
          <div className="text-gray-500">{occupation}</div>
          <div className="text-gray-500">{company}</div>
          <div className="flex space-x-3 pt-6">
            <SocialIcon kind="mail" href={`mailto:${email}`} />
            <SocialIcon kind="github" href={github} />
            <SocialIcon kind="linkedin" href={linkedin} />
            <SocialIcon kind="twitter" href={twitter} />
          </div>
        </div>
        
        {/* Bio Content */}
        <div className="prose max-w-none pb-8 pt-8 dark:prose-invert xl:col-span-2">
          {children}
        </div>
      </div>
    </div>
  )
}
```

**Features**:
- Profile avatar
- Name, occupation, company
- Social media links
- Bio content from MDX

## Layout Selection

### In Blog Posts

Specify layout in frontmatter:

```yaml
---
title: 'My Post'
date: '2024-03-08'
layout: PostLayout  # or PostSimple, PostBanner
---
```

### Default Layout

If no layout specified, defaults to `PostLayout`.

### In Dynamic Pages

```tsx
// app/blog/[...slug]/page.tsx
import { PostLayout, PostSimple, PostBanner } from '@/layouts'

const layouts = {
  PostLayout,
  PostSimple,
  PostBanner,
}

export default async function Page({ params }) {
  const post = allBlogs.find((p) => p.slug === params.slug.join('/'))
  const Layout = layouts[post.layout || 'PostLayout']
  
  return <Layout content={post}>{post.body.code}</Layout>
}
```

## Creating Custom Layouts

<Steps>
  <Step title="Create Layout File">
    Create new layout in `layouts/` directory:
    
    ```tsx
    // layouts/MyCustomLayout.tsx
    import { ReactNode } from 'react'
    import { CoreContent } from 'pliny/utils/contentlayer'
    import type { Blog } from 'contentlayer/generated'
    
    interface LayoutProps {
      content: CoreContent<Blog>
      children: ReactNode
    }
    
    export default function MyCustomLayout({ content, children }: LayoutProps) {
      const { title, date } = content
      
      return (
        <article>
          <header>
            <h1>{title}</h1>
            <time>{date}</time>
          </header>
          <div>{children}</div>
        </article>
      )
    }
    ```
  </Step>
  
  <Step title="Export from Index">
    Add to layout exports (if using index file):
    
    ```tsx
    // layouts/index.tsx
    export { default as MyCustomLayout } from './MyCustomLayout'
    ```
  </Step>
  
  <Step title="Use in Posts">
    Reference in blog post frontmatter:
    
    ```yaml
    ---
    title: 'Test Post'
    layout: MyCustomLayout
    ---
    ```
  </Step>
</Steps>
